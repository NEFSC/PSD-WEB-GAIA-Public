// Helper functions for status display function showStatus(element, message, isError = false) { element.style.display = 'block'; element.className = `usa-alert usa-alert--${isError ? 'error' : 'success'}`; element.innerHTML = `

${message}

`; } // Helper function to convert base64url to ArrayBuffer function base64urlToArrayBuffer(base64url) { const padding = '='.repeat((4 - base64url.length % 4) % 4); const base64 = base64url.replace(/-/g, '+').replace(/_/g, '/') + padding; const rawData = window.atob(base64); const outputArray = new Uint8Array(rawData.length); for (let i = 0; i < rawData.length; ++i) { outputArray[i] = rawData.charCodeAt(i); } return outputArray.buffer; } // Helper function to convert ArrayBuffer to base64url function arrayBufferToBase64url(buffer) { const bytes = new Uint8Array(buffer); let str = ''; for (let i = 0; i < bytes.byteLength; i++) { str += String.fromCharCode(bytes[i]); } const base64 = window.btoa(str); return base64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=/g, ''); } // Initialize WebAuthn functionality document.addEventListener('DOMContentLoaded', function() { const registerButton = document.getElementById('register-passkey'); const testButton = document.getElementById('test-passkey'); const passkeyNameInput = document.getElementById('passkey-name'); const registrationStatus = document.getElementById('registration-status'); const testStatus = document.getElementById('test-status'); // Register passkey if (registerButton) { registerButton.addEventListener('click', async function() { const name = passkeyNameInput.value.trim(); if (!name) { showStatus(registrationStatus, 'Please enter a name for your security key.', true); return; } try { registerButton.disabled = true; registerButton.textContent = 'Creating Security Key...'; // Get registration options from server const optionsResponse = await fetch('/account/2fa/webauthn/register/begin/', { method: 'POST', headers: { 'X-CSRFToken': document.querySelector('[name=csrfmiddlewaretoken]').value, 'Content-Type': 'application/json', }, }); if (!optionsResponse.ok) { throw new Error('Failed to get registration options'); } const options = await optionsResponse.json(); // Convert base64url strings to ArrayBuffers const publicKeyCredentialCreationOptions = { challenge: base64urlToArrayBuffer(options.challenge), rp: options.rp, user: { ...options.user, id: base64urlToArrayBuffer(options.user.id), }, pubKeyCredParams: options.pubKeyCredParams, authenticatorSelection: { // Force no PIN requirement - very aggressive settings userVerification: "discouraged", residentKey: "discouraged", requireResidentKey: false // Don't set authenticatorAttachment to allow any authenticator }, timeout: options.timeout, attestation: options.attestation, excludeCredentials: options.excludeCredentials.map(cred => ({ ...cred, id: base64urlToArrayBuffer(cred.id), })), }; // Create credential const credential = await navigator.credentials.create({ publicKey: publicKeyCredentialCreationOptions }); // Convert credential response to JSON const credentialJson = { id: credential.id, rawId: arrayBufferToBase64url(credential.rawId), response: { attestationObject: arrayBufferToBase64url(credential.response.attestationObject), clientDataJSON: arrayBufferToBase64url(credential.response.clientDataJSON), }, type: credential.type, name: name }; // Send to server const registrationResponse = await fetch('/account/2fa/webauthn/register/complete/', { method: 'POST', headers: { 'X-CSRFToken': document.querySelector('[name=csrfmiddlewaretoken]').value, 'Content-Type': 'application/json', }, body: JSON.stringify({ credential: credentialJson, name: name }), }); if (!registrationResponse.ok) { throw new Error('Registration failed'); } const result = await registrationResponse.json(); if (result.success) { showStatus(registrationStatus, result.message || 'Security key registered successfully!'); // Reload page to show the new credential setTimeout(() => { window.location.reload(); }, 2000); } else { throw new Error(result.error || 'Registration failed'); } } catch (error) { console.error('Registration error:', error); showStatus(registrationStatus, `Registration failed: ${error.message}`, true); registerButton.textContent = 'Add Security Key'; } finally { registerButton.disabled = false; } }); } // Test passkey if (testButton) { testButton.addEventListener('click', async function() { try { testButton.disabled = true; testButton.textContent = 'Testing...'; // Get authentication options from server const optionsResponse = await fetch('/account/2fa/webauthn/authenticate/begin/', { method: 'POST', headers: { 'X-CSRFToken': document.querySelector('[name=csrfmiddlewaretoken]').value, 'Content-Type': 'application/json', }, }); if (!optionsResponse.ok) { throw new Error('Failed to get authentication options'); } const options = await optionsResponse.json(); // Convert base64url strings to ArrayBuffers const publicKeyCredentialRequestOptions = { challenge: base64urlToArrayBuffer(options.challenge), timeout: options.timeout, rpId: options.rpId, allowCredentials: options.allowCredentials.map(cred => ({ ...cred, id: base64urlToArrayBuffer(cred.id), })), userVerification: "discouraged", // Force no PIN requirement }; // Get assertion const assertion = await navigator.credentials.get({ publicKey: publicKeyCredentialRequestOptions }); // Convert assertion response to JSON const assertionJson = { id: assertion.id, rawId: arrayBufferToBase64url(assertion.rawId), response: { authenticatorData: arrayBufferToBase64url(assertion.response.authenticatorData), clientDataJSON: arrayBufferToBase64url(assertion.response.clientDataJSON), signature: arrayBufferToBase64url(assertion.response.signature), }, type: assertion.type, }; // Send to server const authResponse = await fetch('/account/2fa/webauthn/authenticate/complete/', { method: 'POST', headers: { 'X-CSRFToken': document.querySelector('[name=csrfmiddlewaretoken]').value, 'Content-Type': 'application/json', }, body: JSON.stringify({ credential: assertionJson }), }); if (!authResponse.ok) { throw new Error('Authentication failed'); } const result = await authResponse.json(); if (result.success) { showStatus(testStatus, result.message || 'Security key test successful!'); } else { throw new Error(result.error || 'Authentication failed'); } } catch (error) { console.error('Test error:', error); showStatus(testStatus, `Test failed: ${error.message}`, true); } finally { testButton.disabled = false; testButton.textContent = 'Test Security Key'; } }); } });
